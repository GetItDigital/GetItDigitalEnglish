\begin{frame}\ftb{\index{Cage rotor}Cage rotor}
	\begin{itemize}
		\item Conductive rods are located in the grooves of the runner plate package.
		\item The rods are connected to each other at the front by short-circuit rings.
		\item Together, these components form a cage (similar to a hamster cage).
		\item The rods are designed at an angle to reduce harmonics in the rotating magnetic field.
	\end{itemize}
		\f{width=0.5\textwidth}{height=0.6\textheight}{Aufbau_Asynchron_Kaefig2}{{\bf Cage of an asynchronous rotor.} Shown with half of the sheet metal package.}
\end{frame}


\begin{frame}	\ftb{Start-up of the ASM}\index{Asynchronmaschine>Anlauf}
	\begin{itemize}
		\item The stator generates a rotating magnetic traveling wave in the stationary part with the angular velocity:
		$\omega_\mathrm{s} = \frac{\omega_0}{p}$.\pause
		\item The stationary rotor experiences a time-varying field with the frequency $\omega_\mathrm{s}$.\pause
		\item The rotating magnetic field penetrates the rotor and induces a voltage with the frequency $\omega_\mathrm{s}$.\pause
		\item The induced voltage results in a current flow, since the rotor windings are short-circuited.\pause
		\item The induced rotor current opposes the change of the stator field perceived by it (Lenz’s law).\pause
		\item The stator field and the rotor current interact via the Lorentz force, leading to the formation of a torque.\pause
		\item The torque acts in the direction of the stator’s rotating magnetic field.\pause
		\item The rotor begins to rotate.

	\end{itemize}
\end{frame}

\begin{frame}	\ftb{Operation of the ASM}    \index{Asynchronmaschine>Funktionsprinzip}
	\begin{itemize}
		\item As the rotor speed increases, the rotor perceives an increasingly slower change of the stator field:
		$\omega_\mathrm{r} = \omega_\mathrm{s} - \omega_\mathrm{mech}$.\pause
		\item With increasing speed, both the magnitude and the frequency of the induced voltage in the rotor decrease.\pause
		\item The magnitude of the torque decreases.\pause
		\item When the rotor and the stator field rotate at the same frequency, synchronous speed is reached.\pause
		\item The rotor windings no longer experience a change in the magnetic field.\pause
		\item The induced current and the torque of the machine become zero.\pause
		\item Due to friction effects, the rotor is slowed down again, leading to the renewed formation of torque.\pause
		\item In the steady-state condition, a speed slightly below the synchronous speed is established.

	\end{itemize}
\end{frame}


\begin{frame}\ftb{Speed/torque characteristic curve}
	\renewcommand{\figurename}{Figure}
	\begin{columns}
		\column[c]{0.3\textwidth}
			\b{Slippage:}
            \s{The speed/torque characteristic curve of an asynchronous machine is non-linear and can be divided into several sections. The characteristic curve is often plotted as shown in Figure \ref{kloss} using the slip $s$. The slip\index{Slip} refers to the percentage deviation of the mechanical rotational speed of the rotor $n_\mathrm{r}$ from the electrical rotational speed $n_\mathrm{s}$ of the supplying network in the stator.}
			\nomenclature[F]{$s$}{Schlupf der Asynchronmaschine\nomunit{$1$}}
			\nomenclature[F]{$\omega_\mathrm{s}$}{Synchrone Drehfrequenz des speisenden Netzes \nomunit{$\frac{1}{\mathrm{s}}$}}
			\nomenclature[F]{$\omega_\mathrm{r}$}{Drehfrequenz des Rotors \nomunit{$\frac{1}{\mathrm{s}}$}}
			\begin{eq}
				s = \frac{\omega_\mathrm{s} - \omega_\mathrm{r}}{\omega_\mathrm{s}}
			\end{eq}

            \s{A slip of $s=1$ ($100\%$) therefore corresponds to the machine being at a standstill or a speed of $n=0$, while a slip of $s=0$ corresponds to mains-synchronous operation, which can never occur with an asynchronous machine, however, as the torque would be zero in this case.}
			\uncover<2->{
				\s{The speed/torque characteristic curve is also known as the Klosssche characteristic curve and can be calculated using the equation \ref{GlKloss} ausgedrückt werden:}
				\b{Kloss's equation}
				\begin{eq}
					M = \frac{2\cdot M_\mathrm{K}}{\frac{s_\mathrm{K}}{s} + \frac{s}{s_\mathrm{K}}}\label{GlKloss}
				\end{eq}
				\nomenclature[F]{$s_\mathrm{K}$}{Kippschlupf der Asynchronmaschine\nomunit{$1$}}
				\nomenclature[F]{$M_\mathrm{K}$}{Kippmoment der Asynchronmaschine \nomunit{$\mathrm{Nm}$}}
			}
			\uncover<3>{

				\s{For large machines, the slip corresponds very closely to the percentage losses of the machine:}
				\b{Slippage corresponds to the percentage losses:}
				\begin{eq}
					\eta \approx 1-s
				\end{eq}
			}

\begin{frame}\ftx{Example: Asynchronous machine}
	\begin{expl}{asynchronous machine}{}
		A three-phase asynchronous motor has the following information on its nameplate:
		\begin{columns}
			\column[c]{0.5\textwidth}
				\begin{itemize}
					\item Rated power: $10\,$kW
					\item Rated speed: $1440\,\frac{\mathrm{U}}{\mathrm{min}}$
				\end{itemize}
			\column[c]{0.5\textwidth}
				\begin{itemize}
					\item Frequency: $50\,$Hz
					\item Breakdown torque: 25\%
				\end{itemize}
		\end{columns}\b{\vspace{0.3cm}}
		Calculate the rated torque, the tipping torque and the starting torque.
		\s{
			\begin{align*}
				P_\mathrm{N} &= M_\mathrm{N}\cdot \omega_\mathrm{N} = M_\mathrm{N}\cdot 2\pi n_\mathrm{N}\\
				M_\mathrm{N} &= \frac{P_\mathrm{N}}{2\pi n_\mathrm{N}} = \frac{10\cdot 10^3\,\mathrm{W}}{2\pi\cdot 1440 \frac{1}{60\,\mathrm{s}}} = 66,31\,\mathrm{Nm}\\
				s_\mathrm{N} &= \frac{\omega_\mathrm{s} - \omega_\mathrm{r}}{\omega_\mathrm{s}}\\
				&=\frac{2\pi \cdot 1500 \,\frac{1}{60\,\mathrm{s}} - 2\pi\cdot 1440\,\frac{1}{60\,\mathrm{s}}}{2\pi \cdot 1500 \,\frac{1}{60\,\mathrm{s}}}\\
				& = 0,04
			\end{align*}
			To calculate the other moments, equation \ref{GlKloss} is calculated for the cases of nominal operation and standstill.
			\begin{align*}
				M &= \frac{2\cdot M_\mathrm{K}}{\frac{s_\mathrm{K}}{s} + \frac{s}{s_\mathrm{K}}}\\
				M_\mathrm{K} &= \frac{M_\mathrm{N}}{2}\cdot\left(\frac{s_\mathrm{K}}{s_\mathrm{N}} + \frac{s_\mathrm{N}}{s_\mathrm{K}}\right)\\
				&= \frac{66,31\,\mathrm{Nm}}{2}\cdot\left(\!\frac{0,25}{0,04} + \frac{0,04}{0,25}\!\right) = 212,54\,\mathrm{Nm}\\
				M_\mathrm{A} &= \frac{2\cdot 212,54\,\mathrm{Nm}}{\frac{0,25}{1} + \frac{1}{0,25}} = 100,02\,\mathrm{Nm}
			\end{align*}
		}
		\b{
		\only<2-3>{
				\begin{align*}
