    \begin{itemize}
        \item[\bf a)] Reading the phase angle values of voltage and current. Describing the behaviour.
        \item[\bf b)] Specifying the period duration, frequency and angular frequency.
        \item[\bf c)] Calculate the instantaneous power at time t = 12.5 ms.
        \item[\bf d)] Calculate the active power and reactive power.
        \item[\bf e)] Calculate the apparent power using the conjugate complex current.
    \end{itemize}
    
}

\Loesung{
    \begin{itemize}

        \item[\bf a)]
              The voltage has a phase angle of $\varphi_\mathrm{u}=0$ and the current has a phase angle of $\varphi_\mathrm{i}=\pi/3$. \\
              The lagging of the current indicates an inductive behaviour.
              
        \item[\bf b)]
              The period T is $50\ ms$. \\
                The frequency is:
              \begin{eq}
                  f=\frac{1}{T}=\frac{1}{50\ ms}= 20\ Hz   \nonumber
              \end{eq}
              The angular frequency is:
              \begin{eq}
                  \omega = 2\pi \cdot f = 2\pi \cdot 20\ Hz = 125,66\ Hz  \nonumber
              \end{eq}
              
        \item[\bf c)]
              The instantaneous power at time t = 12.5 ms is:
              \begin{eqa}
                  p(t) &= u(t) \cdot i(t) = 12\ V \cdot \sin(2\pi\cdot20\ Hz\cdot 0,0125\ s) \cdot 2\ A \cdot \sin(2\pi\cdot20\ Hz\cdot 0,0125\ s+\frac{\pi}{3}) \nonumber \\
                  p(t) &= 12\ W \nonumber
              \end{eqa}
              
        \item[\bf d)]
              The active power is:
              \begin{eqa}
                  P &= U \cdot I \cdot \cos(\varphi) = \frac{\hat{U}}{\sqrt{2}} \cdot \frac{\hat{I}}{\sqrt{2}} \cdot \cos(\varphi)
                  = \frac{12\ V}{\sqrt{2}} \cdot \frac{2\ A}{\sqrt{2}} \cdot \cos(60^o) \nonumber \\
                  P &= 6\ W    \nonumber
              \end{eqa}
              The reactive power is:
              \begin{eqa}
                  Q &= U \cdot I \cdot \sin(\varphi) = \frac{\hat{U}}{\sqrt{2}} \cdot \frac{\hat{I}}{\sqrt{2}} \cdot \sin(\varphi)
                  = \frac{12\ V}{\sqrt{2}} \cdot \frac{2\ A}{\sqrt{2}} \cdot \sin(60^o) \nonumber \\
                  Q &= 10,392\ var    \nonumber
              \end{eqa}
              
        \item[\bf e)]
              The apparent power is:
              \begin{eqa}
                  \underline{S} &= \underline{U} \cdot \underline{I}^* = 
                  \frac{\underline{\hat{U}}}{\sqrt{2}} \cdot \frac{\underline{\hat{I}}^*}{\sqrt{2}} =
                  \frac{12\ V}{\sqrt{2}} \cdot e^{j(2\pi\cdot20\ Hz)} \cdot \frac{2\ A}{\sqrt{2}} \cdot e^{j(2\pi\cdot20\ Hz+\frac{\pi}{3})} \nonumber \\
                  \underline{S} &= 12\ VA \cdot e^{j\frac{\pi}{3}} = 6\ W + j10,392\ var    \nonumber
              \end{eqa}
              
              
    \end{itemize}
    
}
