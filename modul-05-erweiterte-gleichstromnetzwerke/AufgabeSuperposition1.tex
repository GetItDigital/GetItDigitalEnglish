\subsection{Superposition 1}
\Aufgabe{
    The following network is given below.

    \begin{figure}[H]
        \resizebox{0.6\textwidth}{!}{%  
        \input{Tikz/src/A4Super1.tex}
        }\centering                                                           
    \end{figure}

    Using the superposition method, determine the voltage drop across the resistor $R_L$.
    To do this, follow these steps:
        
    \begin{itemize}
            \item[\bf a)]
                Determine the proportion of $U_g$.
            \item[\bf b)]
                Determine the proportion of $I_g$.
            \item[\bf c)]
                Combine the results using the superposition theorem.
    \end{itemize}
    }

\Loesung{
	\begin{itemize}
        \item[\bf a)]
        Proportion of $U_g$:\\
        \begin{figure}[H]
            \resizebox{0.6\textwidth}{!}{%  
            \input{Tikz/src/B4Super1.tex}
            }\centering                                                           
        \end{figure}

        \begin{eq}
            U_\mathrm{RL}(U_\mathrm{g}) = \frac{R_\mathrm{L}}{R_\mathrm{i} + R_\mathrm{L}} \cdot U_\mathrm{g}       \nonumber
        \end{eq}

		\item[\bf b)]
		Proportion of $I_g$:\\
        \begin{figure}[H]
            \resizebox{0.65\textwidth}{!}{%  
            \input{Tikz/src/B4Super1_zwei.tex}
            }\centering                                                           
        \end{figure}

        \begin{eq}
            U_\mathrm{RL}(I_\mathrm{g}) = \frac{R_\mathrm{i} \cdot R_\mathrm{i}}{R_\mathrm{i} + R_\mathrm{L}} \cdot I_\mathrm{g}       \nonumber
        \end{eq}

        \item[\bf c)]
        Superposition principle:\\

        \begin{eq}
            U_\mathrm{RL} = U_\mathrm{RL}(U_\mathrm{g}) + U_\mathrm{RL}(I_\mathrm{g}) = 
            \frac{R_\mathrm{L}}{R_\mathrm{i} + R_\mathrm{L}} \cdot U_\mathrm{g} + 
            \frac{R_\mathrm{i} \cdot R_\mathrm{i}}{R_\mathrm{i} + R_\mathrm{L}} \cdot I_\mathrm{g}       \nonumber
        \end{eq}
	\end{itemize}
}
