\subsection{Switch-on behaviour of an RL series connection}\label{Einschaltverhalten_RL-Serie}
% Übernommen von Prof. Exknowski, FH SWF Hagen, 2024, ET3 (Übung 7, Aufgabe 2), leicht verändert
\Aufgabe{
The switch-on behavior of a series circuit consisting of a resistive resistor $R$ and an inductance $L=100\,\text{mH}$ is to be examined. 
At time $t=0$, a voltage $u=U_0 = 35\,\text{V}$ is applied to the series circuit. For $t \leq 0$, $u=0$. The inductor voltage at time $t_1=3\,\text{ms}$ is known and amounts to $u_L(t_1)=26\,\text{V}$.\\[2ex]
Determine the time constant $\tau$, the resistance $R$, and the voltage $u_R(t=5\,\text{ms})$.
}

\Loesung{%

    \begin{align*}
        \text{ODE:}&&\quad\frac{L}{R}\cdot\dt i+i&=\frac{U_0}{R}\\
        \text{Solution:}&&i&=\frac{U_0}{R}\cdot(1-e^{-\frac{t}{\tau}})\quad\text{with}\quad \tau = \frac{L}{R}
    \end{align*}
    \begin{align*}
        u_L &= L\cdot\dt i =U_0\cdot e^{-\frac{t}{\tau}}\\
        u_R &= u-u_L = U_0\cdot(1-e^{-\frac{t}{\tau}})\\
        i(t=3\,ms)&= \frac{U_0}{R}\cdot(1-e^{-\frac{t}{\tau}})\\
        u_L(t=3\,ms)&= U_0\cdot e^{-\frac{t}{\tau}} = 26\,V\\
        \frac{u_L(t=3\,ms)}{U_0}&=e^{-\frac{t}{\tau}}\\
        -\frac{t}{\tau}&=\ln\left(\frac{u_L(t=3ms)}{U_0}\right)\\
        \tau&=-\frac{t}{\ln\left(\frac{u_L(t=3ms)}{U_0}\right)} = -\frac{3ms}{\ln\left(\frac{26\,V}{35\,V}\right)} = 10,1\,ms\\
        R &= \frac{L}{\tau} = \frac{100\,mH}{10,1\,ms} = 9,9\,\Omega\\
        u_R(t=5\,ms)&=U_0\cdot(1-e^\frac{-5\,ms}{10,1\,ms})\\
        &=35\,V\cdot 0,39 = 13,65\,V
    \end{align*}

\subsection{ET3 Ü8 Switching behaviour of an RC network}\label{AufgSchaltverhaltenRCNetzwerk}
\Aufgabe{
    % Übernommen von Prof. Exknowski, FH SWF Hagen, 2024, ET3 (Übung 8), leicht verändert

        \\
        
        \begin{tabular}{lll}

            & $C_1 = 1\,\mu \text F$ & $R_3 = 500\,\Omega$	\\
            & $U_q = 250\,\text{V}$	& \\
        \end{tabular}
        \\
        
        Determine the time course of the current $i_3$ through the resistor $R_3$ and plot it as a function of time (line graph) for $-\tau<t<5\tau$. \\
        
        The network is in a steady state at time $-\tau$. Here, $\tau$ is the time constant of the network.
    \end{minipage}

    \begin{minipage}{0.38\textwidth}
        \centering
        \includegraphics{Tikz/pdf/Uebungen/circ_uebung_8.pdf}

\Loesung{
Set up the ODE for $\ecv{u_C}$ and determine the current $\eci{i_3}$ via $\eci{i_3}=\frac{\ecv{u_C}}{R_3}$.\\
Combine $R_1$ and $R_2$ as a parallel connection $R_{12}$ after the switching instant ($t>0$).

\textbf{1st ODE} for $\ecv{u_C}$ ($t > 0$) with $\eci{i_3}=\frac{\ecv{u_C}}{R_3}$: 
\begin{align*}
    \ecv{U_q} &= \ecv{u_{12}} + \ecv{u_C} &&\text{with}\quad \ecv{u_{12}} \mathrlap{\text{ via parallel connection of $R_1$ and $R_2$}}\\
    &= R_{12} \cdot \eci{i_{12}} + \ecv{u_C} &&\text{with}\quad R_{12}=R_1||R_2,& \eci{i_{12}}&=\eci{i_C}+\eci{i_3}\\
    &= R_{12} \cdot (\eci{i_C} + \eci{i_3}) + \ecv{u_C} &&\text{with}\quad \eci{i_{C}}=C\cdot\dt\,\ecv{u_C},& \eci{i_3}&=\frac{\ecv{u_C}}{{R_3}}\\
    &= R_{12} \cdot (C \cdot \dt\,\ecv{u_C} + \frac{\ecv{u_C}}{{R_3}}) + \ecv{u_C}\\
    &= C \cdot R_{12} \cdot \dt\,\ecv{u_C} + \left( \frac{R_{12}}{{R_3}} + 1 \right) \cdot \ecv{u_C}& &\bigg|\ \cdot R_3\\
    \ecv{U_q} \cdot R_3
    &= C \cdot R_{12} \cdot R_3 \cdot \dt\,\ecv{u_C} + \left(R_{12}+R_3\right) \cdot \ecv{u_C}& &\bigg| :\left(R_{12}+R_3\right)\\
    \ecv{U_q} \cdot \frac{R_3}{R_{12}+R_3}
    &= \underbrace{C \cdot \frac{R_{12} \cdot R_3}{R_{12} + R_3}}_{\tau} \cdot \dt\,\ecv{u_C} + \ecv{u_C}& &\Longrightarrow\quad \tau=C \cdot \frac{R_{12} \cdot R_3}{R_{12} + R_3} 
\end{align*}
\textbf{3. Homogeneous solution} (fleeting) and \textbf{2. Particular solution} ($t \to \infty$, settled in):
\begin{align*}
    \ecv{u_{C,\mathrm{h}}} &= K \cdot \mathrm{e}^{-\frac{t}{\tau}}& &\text{with}\quad \tau=C \cdot \frac{R_{12} \cdot R_3}{R_{12} + R_3} \\
    \ecv{u_{C,\mathrm{p}}} &= \ecv{U_q} \cdot \frac{R_3}{R_{12}+R_3}& &\text{$C_1$ corresponds to idle}
\end{align*}
\textbf{4. superposition} and \textbf{5. Determine constant(s)}:
\begin{align*}
    \ecv{u_C} &= \ecv{u_{C,\mathrm{h}}} + \ecv{u_{C,\mathrm{p}}}\\
    \ecv{u_C}(t=0) &= K \cdot \redcancel{\mathrm{e}^{0}} + \ecv{U_q} \cdot \frac{R_3}{R_{12}+R_3} \overset{!}{=} \ecv{U_q}& &\text{(Initial condition)}\\
    \Rightarrow K &= \ecv{U_q} \cdot \left( 1 - \frac{R_3}{R_{12}+R_3} \right)\\
    &= \ecv{U_q} \cdot \frac{R_{12}}{R_{12}+R_3}\\[2pt]
    \ecv{u_C} &= \ecv{U_q} \cdot \left( 1 - \frac{R_3}{R_{12}+R_3} \right) \cdot \mathrm{e}^{-\frac{t}{\tau}} + \ecv{U_q} \cdot \frac{R_3}{R_{12}+R_3}& &\bigg| :R_3\\[2pt]
    \eci{i_3} &= \ecv{U_q} \cdot \left( \frac{1}{R_3} - \frac{1}{R_{12}+R_3} \right) \cdot \mathrm{e}^{-\frac{t}{\tau}} + \ecv{U_q} \cdot \frac{1}{R_{12}+R_3}

\subsection{Switching process for alternating voltage}\label{AufgWechselspannungSchalten}
\Aufgabe{
    At time $t=t_0$, the alternating voltage $u_1(t)$ is applied to 
    the entire circuit consisting of $R_1$ in series with the parallel connection of $R_2$ and $C$.
    For $t<t_0$, the circuit is in a steady state.
    The source voltage is given by $u_1(t) = \hat{U}_1 \cdot \sin(\omega t)$.    
    \begin{figure}[H]\centering
        \includegraphics{Tikz/pdf/Uebungen/circ_uebung_einschaltverhalten_rs_rlp.pdf}
    \end{figure}
    \begin{itemize}
    \item[a)] What current $i_{L,e}(t)$ occurs in the steady state at the inductance? \\
    Determine the amplitude and zero phase angle of the current.
    \item[b)] Set up the ODE of $i_L(t)$ for $t \geq t_0$.
    What is the time constant for the transient state?
    Note (ODE):\qquad $\tau \cdot \frac{\mathrm{d}}{\mathrm{d}t}\, i_L + i_L = \text{disturbance}$
    \item[c)] When must switching occur in order to minimise the transient state of $i_L(t)$?
    At which possible switching times does this become maximum?
    Give the answer using the phase shift $\varphi$ between $u_1(t)$ and $i_L(t)$.
    \item[d)] At the switching instant $t_0$, let the steady-state value of $i_{L,e}(t)$ be at its negative peak. 
    The time constant $\tau$ equals one period $T=\frac{2\pi}{\omega}$.  
    Determine the current maximum during the transient response.
    \end{itemize}
}
\Loesung{
    \begin{itemize}
    \item[a)] Steady state ($t \to \infty$) with complex alternating current calculation:
    \begin{align*}
        \eci{\underline{I}_{L,e}} &= \frac{\ecv{\underline{U}_L}}{\underline{Z}_L}
        = \frac{\ecv{\underline{U}_1}}{\underline{Z}_L} \cdot \frac{R_2 || \underline{Z}_L}{R_1 + R_2 || \underline{Z}_L} 
        \qquad\text{with}\quad
        \ecv{\underline{U}_1} = \ecv{\hat{U}_1} \cdot \mathrm{e}^{\mathrm{j}0} = \ecv{\hat{U}_1}\\
        &= \frac{\ecv{\hat{U}_1}}{\underline{Z}_L} \cdot \frac{\frac{R_2\cdot\underline{Z}_L}{R_2+\underline{Z}_L}}{R_2 + \frac{R_1\cdot\underline{Z}_L}{R_2+\underline{Z}_L}}
        \qquad \bigg| \cdot\frac{R_2+\underline{Z}_L}{R_2+\underline{Z}_L}\\
        &= \frac{\ecv{\hat{U}_1}}{\redcancel{\underline{Z}_L}} \cdot \frac{R_2\cdot\redcancel{\underline{Z}_L}}{R_1\cdot(R_2+\underline{Z}_L)+R_2\cdot\underline{Z}_L}\\
        &= \ecv{\hat{U}_1} \cdot \frac{R_2}{R_1 \cdot R_2 + \mathrm{j}\omega L \cdot(R_1+R_2)}  \\
        \varphi &= 0 + \redcancel{\arctan\left(\frac{0}{R_2}\right)} - \arctan\left(\frac{\omega L \cdot(R_1+R_2)}{R_1 \cdot R_2}\right) \\
        &= - \arctan\left(\frac{\omega L \cdot(R_1+R_2)}{R_1 \cdot R_2}\right),\qquad -\frac{\pi}{2} \leq \varphi \leq 0 \\
        \eci{\hat{I}_{L,e}} &= \ecv{\hat{U}_1} \cdot \frac{R_2}{\sqrt{\left(R_1\cdot R_2\right)^2 + \left(\omega L\cdot (R_1+R_2)\right)^2}}\\
        \eci{i_{L,e}}(t) &= \eci{\hat{I}_{L,e}} \cdot \sin(\omega t + \varphi) \qquad \text{current lags }\ecv{u_1}\text{ with }\varphi<0 
    \end{align*}
    \item[b)] ODE for $i_L(t)$ ($t \geq t_0$):
    \begin{align*}
        \ecv{u_1} &= \ecv{u_{R_1}} + \ecv{u_L} \\
        &= R_1 \cdot (\eci{i_2} + \eci{i_L}) + L \cdot \dt\, \eci{i_L}\\
        &= R_1 \cdot \left( \frac{\ecv{u_L}}{R_2} + \eci{i_L} \right) + L \cdot \dt\, \eci{i_L}\\
        &= \frac{R_1}{R_2} \cdot L \cdot \dt\, \eci{i_L} + R_1 \cdot \eci{i_L} + L \cdot \dt\, \eci{i_L}  \\
        &= L \cdot \left( \frac{R_1}{R_2} + 1 \right) \cdot \dt\, \eci{i_L} + R_1 \cdot \eci{i_L} \qquad \bigg| \cdot \frac{1}{R_1}\\
        \frac{\ecv{u_1}}{R_1} &= \underbrace{L \cdot \left( \frac{1}{R_1} + \frac{1}{R_2} \right)}_{\ =\tau} \cdot \dt\, \eci{i_L} + \eci{i_L}\\ 
        \tau &= L \cdot \left( \frac{1}{R_1} + \frac{1}{R_2} \right) = \frac{L}{R_1||R_2}
    \end{align*}
    \item[c)] Time points for minimum and maximum volatile state ($t_{min},\ t_{max}$). \\
    No settling if 
    $\eci{i_{L,f}}=0 \Leftrightarrow \eci{i_L}=\eci{i_{L,e}} \Leftrightarrow \eci{i_L}(t_0)=\eci{i_{L,e}}(t_0)$ applies. \\
    
    That is, for a vanishing transient state (no settling process), the following applies:
    \begin{align*}
        %\intertext{D.h. für verschwindenden flüchtigen Zustand (kein Einschwingvorgang) folgt:}
        &\text{IC:}&\eci{i_L}(t_0) = \eci{i_{L,e}}(t_0) &= \eci{\hat{I}_{L,e}} \cdot \sin(\omega t_0 + \varphi) \overset{!}= 0 &
            &\Rightarrow \eci{i_L}(t')=\eci{i_{L,e}}(t') \Leftrightarrow \eci{i_{L,f}}(t')=0\\
        &&\Leftrightarrow \omega t_0 + \varphi &\overset{!}{=} 0 + n\cdot \pi &
            &\text{with} \quad n \in \mathbb{N} \\
        &&t_{min} &= \frac{n\cdot \pi - \varphi}{\omega} &
            &\text{corresponds to zero crossings of $\eci{i_{L,e}}$}\\
        \end{align*}
        And for maximum transient state (switching point offset by $90\degree$), the following applies:
        \begin{align*}
        %\intertext{Und für maximalen flüchtigen Zustand (Schaltzeitpunkt $90\degree$ versetzt) folgt:}
        &&t_{max} &= \frac{n\cdot \pi - \varphi}{\omega} + \frac{T}{4} &
            &\text{corresponds to extreme points of $\eci{i_{L,e}}$}
    \end{align*}
    \item[d)] Maximum current during the transient process sought.
        Switching time $t_0$ at negative peak value of $i_{L,e}$.
        Let $t' = t - t_0$, i.e. $t'=0$ at the switching time, then the following applies:
    \begin{align*}
        \eci{i_L}(t') &= \eci{i_{L,h}}(t') + \eci{i_{L,e}}(t') \\
        &= K \cdot \mathrm{e}^{-\frac{t'}{\tau}} - \eci{\hat{I}_{L,e}} \cdot \cos(\omega t')\\
        \text{AB:}\quad \eci{i_L}(t'=0) &= K - \eci{\hat{I}_{L,e}}  \overset{!}{=}0 \quad \Rightarrow \quad K = \eci{\hat{I}_{L,e}}\\
        \eci{i_L}(t') &= \eci{\hat{I}_{L,e}} \cdot \left( \mathrm{e}^{-\frac{t'}{\tau}} - \cos(\omega t') \right) \\
        \end{align*}
        Maximum at positive peak value ($180\degree$ to $t_0$):
        \begin{align*}
        %\intertext{Maximum bei positivem Scheitelwert ($180\degree$ nach $t_0$):}
        \eci{i_{L,max}} &= \eci{i_L}(t' = \frac{\pi}{\omega}) \\
        &= \eci{\hat{I}_{L,e}} \cdot \left( \mathrm{e}^{-\frac{T/2}{T}} - \cos(\pi) \right) \\
        &= \eci{\hat{I}_{L,e}} \cdot \left( \mathrm{e}^{-\frac{1}{2}} + 1\right) = 1,606 \cdot \eci{\hat{I}_{L,e}}
    \end{align*}
    \end{itemize}
