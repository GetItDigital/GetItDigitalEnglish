\subsection{Three-phase current 1}
\Aufgabe{
    A three-phase motor has the following nominal values:
    \begin{itemize}
        \item Nennspannung $U_N = 400\ V$
        \item Nennstrom $I_N = 10\ A$
    \end{itemize}
 
    \begin{itemize}
        \item [\bf a)] Calculate the voltage across each winding of the motor in star connection $U_\star$.
        \item [\bf b)] Calculate the voltage across each winding of the motor in delta connection $U_\delta$.
        \item [\bf c)] Calculate the current through each winding $I_\star$ in star connection.
        \item [\bf d)] Calculate the current through each winding $I_\Delta$ in delta operation.
    \end{itemize}
}


\Loesung{
	\begin{itemize}

	\item[\bf a)] Voltage across each winding of the motor $U_\Stern$ in star operation:
	\begin{eqa}
        U_\Stern = \frac{U_N}{\sqrt{3}} = \frac{400\ V}{\sqrt{3}} \approx 230.94\ V  \nonumber   
    \end{eqa}
	
	\item[\bf b)] Voltage across each winding of the motor $U_\Dreieck$ in delta operation:
	\begin{eqa}
        U_\Dreieck = U_N = 400\ V  \nonumber
    \end{eqa}

	\item[\bf c)] Current through each winding $I_\Stern$ in star mode: 
	\begin{eqa}
        I_\Stern = I_N = 10\ A  \nonumber
    \end{eqa}

	\item[\bf d)] Current through each winding $I_\Dreieck$ in delta connection:
    \begin{eqa}
        I_\Dreieck = \sqrt{3} \cdot I_\Stern = \sqrt{3} \cdot 10\ A \approx 17.32\ A  \nonumber
    \end{eqa}    


\end{itemize}

}
