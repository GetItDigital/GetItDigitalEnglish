\section{Simple resistor networks}




\begin{frame}{}
	\s{
	
		Electrical networks are often composed of simpler sub-circuits. It is often helpful to identify these sub-circuits,
 		simplify them, and then combine them into the overall circuit. Such combinations of components
		are generally permissible, provided that the \textbf{terminal behaviour}, i.e. the behaviour between current and voltage 
		between the connection points, does not change. 
	}
	
	
	
	\ftx{Lernziele: Einfache Gleichstromsnetzwerke}
	
	\begin{Learning objectives}{Simple resistance networks}
		The students can
		\begin{itemize}
			\item Identify subcircuits in direct current networks
			\item Simplify and summarise resistance networks
			\item Determine short-circuit and no-load data
			\item Apply superposition methods
		\end{itemize}
	\end{Learning objectives}

		\subsection{Series connection of resistors}
		
		In a series connection, the same current $I$ flows through all components 
        (see Figure \ref{fig:reihenschaltung}). 
		
		
		\begin{figure}[h!]
			\renewcommand{\figurename}{Figure}
			\begin{center}
				\input{Tikz/src/3_widerstandsnetz/Reihenschaltung_Wiederstaende_1}
				\caption{Series connection of two resistors and a voltage source}
				
				\label{fig:reihenschaltung}
			\end{center}
		\end{figure}
		
		Applying the mesh rule (here: counterclockwise direction of rotation) leads to the following mesh equation:
		
		\begin{equation*}
			U_0 - U_1 - U_2 =0 
		\end{equation*}
		
        With the help of Ohm's law, the partial voltages $U_1$ and $U_2$ can be represented as the product of current and resistance:

		
		\begin{equation*}
			U_0 - R_1 \cdot I - R_2 \cdot I = 0
		\end{equation*}
		\begin{equation*}
			U_0 - (R_1+R_2) \cdot I = 0
		\end{equation*}
		
		In this example, resistors $R_1$ and $R_2$ can be combined by adding their resistance values
		to $R_1 + R_2 = R_\mathrm{total}$ (Figure \ref{fig:rges}).
		
		
		\begin{figure}[h!]
			\renewcommand{\figurename}{Figure}
			\begin{center}
				\input{Tikz/src/3_widerstandsnetz/Reihenschaltung_Wiederstaende_2}
				
				\caption{Combining $R_1$ and $R_2$ to form $R_\mathrm{total}$}
				
				\label{fig:rges}
			\end{center}
		\end{figure}
		
		In general, the total voltage $U_\mathrm{ges}$ can be summarised according to this principle as follows:
		
		\begin{equation}
			U_\mathrm{ges} = \sum_{k=1}^{n} U_k = \sum_{k=1}^{n} R_k \cdot I = R_\mathrm{ges} \cdot I
			\label{eq:gesamtwiderstand}
		\end{equation}
		
		A comparison of the coefficients of the last two terms of equation \ref{eq:gesamtwiderstand} yields the generally valid 
		result for combining resistors in a series connection:
		
		\begin{Key point}{Total resistance of a series connection}{}
			\begin{equation}
				R_\mathrm{ges} = \sum_{k=1}^{n} R_k
			\end{equation}
		\end{Key point}
		
		If, instead of the resistance values $R$, we consider the conductance values $G = 1/R$ of the resistors, the total conductance
        of a series connection is:

		
		
		\begin{Key point}{Total conductance of a series connection}{}
			
			
			\begin{equation}
				\frac{1}{G_\mathrm{ges}} = \sum_{k=1}^{n} \frac{1}{G_k}
			\end{equation}
			%	\begin{equation}

			%	\end{equation}
			
			
		\end{Key point}

		%	\end{equation*}

		\subsection{Parallel connection of resistors}
		
        In a parallel connection of resistors, the same voltage is applied to all components (see Figure \ref{fig:parallelschaltung}):		
		
		
		\begin{figure}[h!]
			\renewcommand{\figurename}{Figure}

			\label{fig:parallelschaltung}
			
			
		\end{figure}
		
		
		\begin{equation*}
			U_\mathrm{1} = U_\mathrm{2} = U_0
		\end{equation*}
		
		By applying the node rule, it can be shown that the total current $I_0$ before the resistors is divided into the partial currents 
        $I_1$ and $I_2$:
		
		
		%\begin{equation}
		%I_0 - I_1 - I_2 = 0
		%\label{eq:knotenparallel}	
		
		%\end{equation}
		
		\begin{equation}
			I_0 - I_1 - I_2 = 0
			\label{eq:gesamtwiderstand2}
		\end{equation}
		
		
		As with series connection in Section 4.1, it is often advantageous to combine resistors connected in parallel 
        into a total resistance $R_\mathrm{total}$ (see Figure \ref{fig:parallelzusammen}):
		
		\begin{figure}[h!]
			\renewcommand{\figurename}{Figure}
			\begin{center}

				\input{Tikz/src/3_widerstandsnetz/Parallelschaltung_Wiederstaende_2}
				
			\end{center}
			\caption{Summary of the parallel-connected resistors $R_1$ and $R_2$ from Figure \ref{fig:parallelschaltung} to $R_\mathrm{total}$}
			\label{fig:parallelzusammen}
		\end{figure}
		
        With the help of Ohm's law, the unknown partial currents $I_1$ and $I_2$ can be quantified from the node rule (equation \ref{eq:gesamtwiderstand2}):		
		\begin{equation*}
			I_1 = \frac{U_0}{R_1}, \quad I_2 = \frac{U_0}{R_2}
		\end{equation*}
		
		Inserted into equation \ref{eq:gesamtwiderstand2}, this gives:
		
		
		
		\begin{equation*}
			I_0 - \frac{U_0}{R_1} - \frac{U_0}{R_2} = 0
		\end{equation*}
		
		
		\begin{equation*}
			I_0 - \bigg( \frac{1}{R_1}+\frac{1}{R_2} \bigg) \cdot U_0 = 0
		\end{equation*}
		
		\begin{equation*}
			\rightarrow I_0 = U_0 \cdot  \bigg( \frac{1}{R_1}+\frac{1}{R_2} \bigg)
		\end{equation*}
		
		
		A coefficient comparison with Ohm's law applied to Figure \ref{fig:parallelzusammen} 
		( $I_0 = U_0 \cdot \frac{1}{R_\mathrm{ges}}$)
		
		provides for $R_\mathrm{total}$ in the case shown of two resistors connected in parallel:
		
		\begin{equation*}
			\frac{1}{R_\mathrm{ges}} =  \frac{1}{R_1}+\frac{1}{R_2}
		\end{equation*}
		
		Multiplying out this expression gives the form frequently used in calculations:
		
		\begin{equation}
			R_\mathrm{ges} = \frac{R_1\cdot R_2}{R_1 + R_2}
			\label{eq:r2parallel}
		\end{equation}
		
		
		The total resistance of a parallel connection of any number of resistors can be determined using the general formula
		for parallel connections:
		
		\begin{Key point}{Total resistance of a parallel connection}
			
			\begin{equation}
				\frac{1}{R_\mathrm{ges}} = \sum_{k=1}^{n} \frac{1}{R_k}
				\label{eq:rparallel}
			\end{equation}
			%	\begin{equation}

			%	\end{equation}
		\end{Key point}

		parallel connection of resistors is often easier to calculate:
		
		
		
		\begin{Key point}{Total conductance of a parallel connection}
			
			\begin{equation}
				G_\mathrm{ges} = \sum_{k=1}^{n} G_k 
			\end{equation}
			%	\begin{equation}
			%		\frac{1}{R_ges} = \sum_{k=1}^{n} \frac{1}{R_k} 
			%	\end{equation}
		\end{Key point}

	\subsection{Voltage divider on resistors}
	
	\s{
	
		One application of connecting two resistors in series is the voltage divider. It is used to divide an 
		input voltage $U_0$ into two smaller partial voltages, thus providing a precisely defined output voltage $U_2$.
		
		
		
		\subsubsection{The unbalanced voltage divider}
		
		First, the voltage divider shown in Figure \ref{fig:spannungsteiler} is unloaded,
        i.e. no load resistor is connected to the open terminals.
        All components are traversed by the same current $I$:

		\begin{figure}[h!]
			\renewcommand{\figurename}{Figure}
			\begin{center}

				\input{Tikz/src/3_widerstandsnetz/Spannungsteiler_unbelastet}
				\caption{Unloaded voltage divider}
				\label{fig:spannungsteiler}
			\end{center}
		\end{figure}

		\begin{equation*}
			I = \frac{U_1}{R_1} = \frac{U_2}{R_2}
		\end{equation*}
		
		By combining the series connection of resistors, the following result is obtained:
		
		\begin{equation*}
			I = \frac{U_0}{R_1+R_2}
		\end{equation*}
		
		Equating the two voltage terms yields the ratio equation: 
		
		
		
		\begin{equation*}
			\frac{U_2}{R_2} = \frac{U_0}{R_1+R_2}
		\end{equation*}
		
		Isolating the output voltage on the left side of the equal sign ultimately leads 
		to the general equation for the unloaded voltage divider:
		
		\begin{Key point}{Unloaded voltage divider}
			\begin{equation}
				U_2 = U_0 \cdot \frac{R_2}{R_1+R_2}
			\end{equation}

		\end{equation*}

		\end{equation*}

		\end{equation*}

			U = U_0 - U_\mathrm{Ri}
		\end{equation*}

		\begin{equation*}
			U_0 << I \cdot R_\mathrm{i}
		\end{equation*}

		\end{equation*}

		\begin{Key point}{Open-circuit voltage of an equivalent voltage source}{}
			\begin{equation*}
				U_0 = U_\mathrm{L}
			\end{equation*}
		\end{Key point}
		
		In the short-circuit test, the two terminals of the active two-pole are directly short-circuited, i.e. an
		ideally conductive connection is established between them (Figure \ref{fig:kurzschluss1}).

		No output voltage $U$ can be applied across the short circuit thus created:
		
		\begin{equation*}
			U = 0
		\end{equation*}
		
		Consequently, the entire voltage generated by $U_0$ drops across the internal resistance $R_\mathrm{i}$.
		According to Ohm's law, this determines the short-circuit current $I_\mathrm{K}$. 
		
		\begin{equation*}
			U_\mathrm{Ri} = U_\mathrm{L} =R_\mathrm{Ri} \cdot I_\mathrm{K}
		\end{equation*}
		
        The short-circuit current $I_\mathrm{K}$ is therefore:		
		
		
		
		\begin{equation*}
			I_\mathrm{K} = \frac{U_\mathrm{L}}{R_\mathrm{i}}
		\end{equation*}
		
        The internal resistance $R_\mathrm{i}$ can therefore be calculated as follows:

		\begin{Key point}{Internal resistance of an equivalent voltage source}{}
			
			\begin{equation*}
				R_\mathrm{i} = \frac{U_\mathrm{L}}{I_\mathrm{K}}
			\end{equation*}
			
		\end{Key point}
		
		\textbf{Attention:} $U_\mathrm{L}$ und $I_\mathrm{K}$ are certain quantities in two completely independent experiments that 
        do not occur simultaneously. The formula therefore only looks like Ohm's law at resistance $R_\mathrm{i}$,
        but it is not.

		\paragraph{The real power source}\\
		
		\phantom{text}\\
		
		
		
		
		A \textbf{real current source} also consists of an 
        ideal current source $U_0$ and an internal resistance $R_\mathrm{i}$
        (see Figure \ref{fig:stromq}). However, in contrast to the real voltage source, this is 
        connected in parallel rather than in series with the source.
		
		
		
		\begin{figure}[h!]
			\renewcommand{\figurename}{Figure}		
			\centering
			\input{Tikz/src/3_widerstandsnetz/reale_Stromquelle}
			\caption{Real current source consisting of an ideal current source $I_0$ and an internal resistance $R_\mathrm{i}$, which is loaded with a resistance $R_\mathrm{a}$}
			\label{fig:stromq}
			
		\end{figure}
		
		
		Depending on the output voltage $U$ that occurs, the current $I_\mathrm{Ri} = U / R_\mathrm{Ri}$ flows through the finite internal resistance 
        $R_\mathrm{i}$, so the output current $I$ is 
		given by
		
		\begin{Key point}{Output current of a real current source}
			
			\begin{equation*}
				I = I_0 - \frac{U}{R_\mathrm{i}}
			\end{equation*}
			
		\end{Key point}
		
        In order to reduce the output current $I$ through the internal resistance, this should be selected as large as possible:
		
		
		\begin{equation*}
			I_0 >> \frac{U}{R_\mathrm{i}}
		\end{equation*}
		
		
		
	}

	\s{
	
		\paragraph{The backup power source}\\
		\phantom{text}\\
		
		
		
		
		As an alternative to the replacement voltage source, an active two-pole can also be represented as a replacement current source
		(Figure \ref{fig:ersatzstromquelle2}). It can also be fully characterised by an ideal current source $I_0$ and a combined 
		internal resistance $R_\mathrm{i}$.
		
		\begin{figure}[h!]
			\renewcommand{\figurename}{Figure}
			\begin{center}

				\input{Tikz/src/3_widerstandsnetz/Ersatzstromquelle}
				
			\end{center}
			
			\caption{The replacement current source (dashed box) has the same equivalent circuit diagram as a real current source.}
			\label{fig:ersatzstromquelle2}
		\end{figure}
		
		Similar to the equivalent voltage source, the parameters $I_0$ and $R_\mathrm{i}$ can also be determined here using the 
        open-circuit and short-circuit tests.  
        
		In the \textbf{open-circuit test}, the entire current $I_0$ flows through the internal resistance. The following open-circuit voltage $U_\mathrm{L}$ is established at the terminals connected in parallel:
		
		\begin{equation*}
			U_\mathrm{L} = R_\mathrm{i} \cdot I_0
		\end{equation*}
		
		
		\begin{figure}[h!]
			\renewcommand{\figurename}{Figure}

	\s{
		In network analysis, it can be useful to convert current sources into voltage sources or
		voltage sources into current sources. This works identically for real current and voltage sources,
		as well as for equivalent current and voltage sources. The value of the internal resistance 
		$R_\mathrm{i}$ remains the same, but its position in the equivalent circuit diagram changes (see Figure \ref{fig:umwandlung})
		
		\begin{figure}[h!]
			\renewcommand{\figurename}{Figure}
			\begin{center}

		and are converted to the other source.
		
		\textbf{Attention}: The counting arrows of current and voltage sources converted into each other are directed in opposite directions!

		\begin{Key point}{Voltage of a converted voltage source}{}

		\begin{Key point}{Current from a converted current source}{}

		\end{Key point}
