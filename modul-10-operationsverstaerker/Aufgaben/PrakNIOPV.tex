\subsection{Investigation of the amplification of an OP with variable resistors\label{Aufg24}}
% Alt: Praktikumsaufgabe 10
\Aufgabe{
    Investigate the amplification of a non-inverting operational amplifier with variable resistors and calculate the output voltage $U_\mathrm{A}$.
    Given:
    \begin{itemize}
        \item Supply voltage: $\pm 15\,\mathrm{V}$
        \item Input voltage: $U_\mathrm{E} = 1\,\mathrm{V}$
        \item Feedback resistance: $R_\mathrm{R} = 100\,\Omega$
        \item Input resistance: $R_\mathrm{E} = 680\,\Omega$
        \item Further possible resistance values: $R_\mathrm{R} = 220\,\Omega$, $R_\mathrm{E} = 680\,\Omega$
        \item Load resistance: $R_\mathrm{Last} = 10\,\mathrm{k\Omega}$
    \end{itemize}
    
    \begin{itemize}
        \item[a)] Calculate the gain $V$ and the output voltage $U_\mathrm{A}$ for the given resistor values.  
        \item[b)] How does the gain change if $R_\mathrm{R}$ is increased from $100\,\Omega$ to $220\,\Omega$?  
        \item[c)] What effects does a change in $R_\mathrm{E}$ have on the gain and the output voltage?  
        \item[d)] What limitations for $U_\mathrm{A}$ result from the supply voltage?

    \end{itemize}

    \end {figure}
}

\Loesung{
        \textbf{a) Calculation of the output voltage $U_\mathrm{A}$ for given values} \\
        Given is:
        \[
        U_\mathrm{E} = 1\,\mathrm{V}, \quad R_\mathrm{R} = 100\,\Omega, \quad R_\mathrm{E} = 680\,\Omega
        \]
        Insert into the amplification formula:
        \[
        V = 1 + \frac{100\,\Omega}{680\,\Omega} = 1 + 0{,}147 = 1{,}147
        \]
        The output voltage is calculated as follows:
        \[
        U_\mathrm{A} = V \cdot U_\mathrm{E} = 1{,}147 \cdot 1\,\mathrm{V} = 1{,}147\,\mathrm{V}
        \]
    
        \textbf{b) Change from $R_\mathrm{R}$ to $220\,\Omega$} \\[0.2cm]
        When $R_\mathrm{R} = 220\,\Omega$ is selected:
        \[
        V = 1 + \frac{220\,\Omega}{680\,\Omega} = 1 + 0{,}323 = 1{,}323
        \]
        \[
        U_\mathrm{A} = 1{,}323 \cdot 1\,\mathrm{V} = 1{,}323\,\mathrm{V}
        \]
    
        \textbf{c) Influence of $R_\mathrm{E}$ on the gain} \\[0.2cm]
        An increase in $R_\mathrm{E}$ leads to a smaller gain $V$, while a decrease in $R_\mathrm{E}$ increases the gain.  
        In practice, $R_\mathrm{E}$ should be chosen so that the output signal does not fluctuate too much.
    
        \textbf{d) Limitation of $U_\mathrm{A}$ by the supply voltage} \\[0.2cm]
        The output voltage $U_\mathrm{A}$ can never exceed the supply voltage $\pm 15\,\mathrm{V}$.
        If $U_\mathrm{A}$ would exceed this range according to calculations, the output voltage is limited to a maximum of $15\,\mathrm{V}$.
        See section: \ref{fig: Verschaltung Verstaerker} and Table \ref{tab:Grundschaltungen}
}
