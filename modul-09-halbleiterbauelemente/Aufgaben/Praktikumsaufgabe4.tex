\subsection{Internship task 4\label{Aufg32}}
\Aufgabe{

    
    Calculate the output voltage for different input voltages of a bridge rectifier.
    
    Gegeben:
    \begin{itemize}
        \item AC Voltage $\mathrm{U_E}$ with amplitudes of 0{,}5\,V, 1\,V, 2\,V, 3\,V
        \item Four idealised diodes (with 0{,}7\,V Forward voltage)
    \end{itemize}
    
    \begin{itemize}
        \item[a)] Determine the output voltage for each input variable.
        \item[b)] Explain the input voltage at which the bridge rectifier begins to function.
    \end{itemize}
}
