\subsection{Description of the ideal characteristics of an operational amplifier\label{Aufg8}}
% Alt: Aufgabe 8
\Aufgabe{
    Describe the following characteristics of an ideal operational amplifier:
    \begin{itemize}
        \item [ \bf a)] Input resistance
        \item [ \bf b)] Output resistance
        \item [ \bf c)] Differential amplification
        \item [ \bf d)] Input offset voltage
    \end{itemize}

\Loesung{
    \begin{itemize}
        \item [ \bf a)] Input resistance: infinite, to minimize loading on the voltage source  
        \item [ \bf b)] Output resistance: $0\,\Omega$, so that the op-amp can deliver as much power as possible  
        \item [ \bf c)] Differential gain: infinite, to amplify small signals  
        \item [ \bf d)] Input offset voltage: $0\,\mathrm{V}$, so that the op-amp produces no output voltage when there is no input signal
    \end{itemize}
