\subsection{Mesh analysis 1}
\Aufgabe{
    Given is the network of a real current source $I_\mathrm{q} = 6\ A$ with a conductance
    vo $G_\mathrm{i} = \frac{2}{3}\ S$.

    \begin{figure}[H]
        \resizebox{0.3\textwidth}{!}{%  
        \input{Tikz/src/A7Masch1.tex}  
        }\centering                                                           
    \end{figure}

    Convert the specified real current source into a real voltage source.
    To do this, follow these steps:
        
    \begin{itemize}
            \item[\bf a)]
                Determine the resistance value of the conductance. 
            \item[\bf b)]
                Convert the real current source into a real voltage source. 
            \item[\bf c)]
                Calculate the voltage of the converted voltage source.
    \end{itemize}
    }

\Loesung{
	\begin{itemize}
        \item[\bf a)]
            Resistance value:    \\

            \begin{eq}
                R_\mathrm{i} = \frac{1}{G_\mathrm{i}} = \frac{1}{\frac{2}{3}\ S} = 1,5\ \Omega   \nonumber
            \end{eq}

		\item[\bf b)]
		    Voltage source:    \\

            \begin{figure}[H]
                \resizebox{0.3\textwidth}{!}{%  
                \input{Tikz/src/B7Masch1.tex} 
                }\centering                                                           
            \end{figure}

        \item[\bf c)]
            Voltage value \\

            \begin{eq}
                U_\mathrm{q} = R_\mathrm{i} \cdot I_\mathrm{q} = 1,5\ \mathrm{\Omega} \cdot 6\ \mathrm{A} = 9\ \mathrm{V}   \nonumber
            \end{eq}

	\end{itemize}
}
