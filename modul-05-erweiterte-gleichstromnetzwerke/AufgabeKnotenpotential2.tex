\subsection{Nodal analysis method 2}
\Aufgabe{
    The following network shows two parallel equivalent circuits of battery cells.
    They consist of voltage sources $U_1$ and $U_2$ as well as internal resistances $R_{i1}$ and $R_{i2}$.
    Perform the analysis using the node potential method. 

    \begin{center}
        \input{Tikz/src/A6Knot2.tex}
    \end{center}

    \begin{enumerate}
        \item[\bf a)] Rearrange the network so that the node potential method can be applied.  
        \item[\bf b)] Set up the bus admittance matrix.
        \item[\bf c)] Set up the vector of the node inflows (I).
        \item[\bf d)] Solve the system of equations and calculate the voltage $U_0$ between terminals $A$ and $B$.
    \end{enumerate}
    }

\Loesung{
	\begin{itemize}
        \item[\bf a)]
            Conversion:

            \begin{figure}[H]
                \input{Tikz/src/B6Knot2.tex}
                \centering
            \end{figure}

            \begin{eq}
                I_1 = \frac{U_1}{R_\mathrm{i1}} = \frac{0,8\ V}{4,8\ \Omega} = 166\ mA      \qquad
                I_2 = \frac{U_2}{R_\mathrm{i2}} = \frac{1,5\ V}{0,8\ \Omega} = 1,875\ A      \nonumber
            \end{eq}

            \begin{eq}
                G_\mathrm{i1} = \frac{1}{R_\mathrm{i1}} = \frac{1}{4,8\ \Omega} = 208\ m\Omega      \qquad
                G_\mathrm{i2} = \frac{1}{R_\mathrm{i2}} = \frac{1}{0,8\ \Omega} = 1,25\ \Omega      \nonumber
            \end{eq}

		\item[\bf b)]
            Bus admittance matrix:

            \begin{eq}
                KAM = (G_\mathrm{i1} + G_\mathrm{i2}) = 208\ m\Omega + 1,25\ \Omega     \nonumber
            \end{eq}

        \item[\bf c)]
            Vector of node inflows: \\

            \begin{eq}
                I = (I_1 + I_2) = 166\ mA + 1,875\ A    \nonumber
            \end{eq}

        \item[\bf d)]
            Solve the system of equations:\\

            \begin{eqa}
                KAM \cdot U &= I     \nonumber\\ 
                (G_\mathrm{i1} + G_\mathrm{i2}) \cdot U_0 &= (I_1 + I_2)     \nonumber\\ 
                (208\ m\Omega + 1,25\ \Omega) \cdot U_0 &= (166\ mA + 1,875\ A)  \nonumber\\ 
                U_0 = \frac{I_1 + I_2}{G_\mathrm{i1} + G_\mathrm{i2}} 
                &= \frac{166\ mA + 1,875\ A}{208\ m\Omega + 1,25\ \Omega} 
                = 1,4\ V    \nonumber
            \end{eqa}

	\end{itemize}
}
