\subsection{Nodal analysis method 2}
\Aufgabe{
    Consider the following network consisting of voltage sources \(U_1\), \(U_2\), \(U_3\) and resistors \(R_1, R_2, R_3, R_4, R_6, R_5\). 
    Perform the analysis using the node potential method. The parameters are given as follows:

    - Voltage sources:
    \[ U_1 = 2 \, \text{V}, \quad U_2 = 2 \, \text{V}, \quad I_3 = 1 \, \text{A} \]

    - Resistances:
    \[ R_1 = R_2 = R_4 = R_6 = R_5 = 2R \quad R_3 = R \quad \text{mit } R = 1 \, \Omega \]

    \begin{center}
        \begin{circuitikz} 
            \draw (1,4) to[R, l=$R_3$, european,*-*] (4,4)
            to[R, l=$R_6$, european,-*] (7,4) -- (9,4)
            (9,4) to[R, l=$R_5$, european,-*] (9,0)
            (11,0) to[I, l_=$I_3$] (11,4)

            (1,4) to[R, l=$R_2$, european,-*] (1,0)
            (4,4) to[R, l_=$R_4$, european] (4,2)
            (4,2) to[V, l=$U_2$, -*] (4,0)

            (-2,0) to[V, l=$U_1$] (-2,2)
            (-2,2) to[R, l=$R_1$, european] (-2,4) -- (1,4)
    
            (9,0) -- (-2,0);
        \end{circuitikz}
    \end{center}

    \begin{enumerate}
        \item[\bf a)] Formen Sie das Netzwerk um, sodass das Knotenpotentialverfahren anwendbar wird.  
            Identifizieren Sie den Masseknoten und die relevanten Knotenpotentiale.
        \item[\bf b)] Stellen Sie das Gleichungssystem für die Knotenpotentiale auf.  
            Nutzen Sie die Leitwerte \( G = \frac{1}{R} \), die aus den Widerständen berechnet werden können, und stellen Sie die Matrixdarstellung der Gleichungen auf.
        \item[\bf c)] Berechnen Sie die Knotenpotentiale.  
            Bestimmen Sie die Potentiale \(U_{10}, U_{20}, U_{30}\) durch Lösung des Gleichungssystems.
        \item[\bf d)] Berechnen Sie den Strom \(I_4\) durch den Widerstand \(R_4\). 
            Verwenden Sie die berechneten Knotenpotentiale und die Beziehung:
            \[ I_4 = \frac{U_{20} - U_{30}}{R_4} \]
    \end{enumerate}
    }

\Loesung{
	\begin{itemize}
        \item[\bf a)]
            Prepare network for the nodal analysis method:\\
            - We select the lower node as the ground node. (\( U = 0 \)).\\
            - The relevant node potentials are:\\
            \qquad $U_{10}$ \quad Potential at $R_1$, $R_2$, $R_3$\\
            \qquad $U_{20}$ \quad Potential at $R_4$, $R_6$, $R_5$\\
            \qquad $U_{30}$ \quad Potential at $U_3$, $R_5$.

            \begin{center}
                \begin{circuitikz} 
                    \draw (1,4) to[R, l=$R_3$, european,*-*] (4,4)
                    to[R, l=$R_6$, european] (7,4) 
                    to[R, l=$R_5$, european] (9,4)
                    (9,0) to[V, l_=$U_3$] (9,4)
        
                    (1,4) to[R, l=$R_2$, european,-*] (1,0)
                    (4,4) to[R, l_=$R_4$, european] (4,2)
                    (4,2) to[V, l=$U_2$, -*] (4,0)
        
                    (-2,0) to[V, l=$U_1$] (-2,2)
                    (-2,2) to[R, l=$R_1$, european] (-2,4) -- (1,4)
            
                    (9,0) -- (-2,0);
                \end{circuitikz}
            \end{center}

		\item[\bf b)]
            System of equations for the node potentials:\\
            The conductance matrix is as follows:
            \begin{eq}
                G =
                \begin{pmatrix}
                \frac{1}{R_1} + \frac{1}{R_2} + \frac{1}{R_3} & -\frac{1}{R_3} & 0 \\
                -\ \frac{1}{R_3} & \frac{1}{R_3} + \frac{1}{R_4} + \frac{1}{R_6} & -\frac{1}{R_6} \\
                0 & -\frac{1}{R_6} & \frac{1}{R_6} + \frac{1}{R_5}
                \end{pmatrix}   \nonumber
            \end{eq}

            Insertion of resistance values (\(R_1 = R_2 = R_4 = R_6 = R_5 = 2R, R_3 = R\)):
            \begin{eq}
                G =
                \begin{pmatrix}
                \frac{1}{2R} + \frac{1}{2R} + \frac{1}{R} & -\frac{1}{R} & 0 \\
                -\frac{1}{R} & \frac{1}{R} + \frac{1}{2R} + \frac{1}{2R} & -\frac{1}{2R} \\
                0 & -\frac{1}{2R} & \frac{1}{2R} + \frac{1}{2R}
                \end{pmatrix}   \nonumber
            \end{eq}

            With \(R = 1 \, \Omega\) will be:
            \[
            G =
            \begin{pmatrix}
            2 & -1 & 0 \\
            -1 & 2 & -0,5 \\
            0 & -0,5 & 1
            \end{pmatrix}
            \]

            The current vector is:
            \[
            I_Q =
            \begin{pmatrix}
            \frac{U_1}{R_1} \\
            \frac{U_2}{R_4} \\
            U_3
            \end{pmatrix}
            =
            \begin{pmatrix}
            1 \\
            \frac{1}{3} \\
            \frac{1}{3}
            \end{pmatrix}
            \]

            The system of equations is:
            \begin{eq}
                G \cdot U = I_\mathrm{Q} \nonumber
            \end{eq}

        \item[\bf c)]
            Calculating node potentials: \\
            Solving the system of equations:
            \[
            \begin{pmatrix}
            2 & -1 & 0 \\
            -1 & 2 & -0,5 \\
            0 & -0,5 & 1
            \end{pmatrix}
            \begin{pmatrix}
            U_{10} \\
            U_{20} \\
            U_{30}
            \end{pmatrix}
            =
            \begin{pmatrix}
            1 \\
            \frac{1}{3} \\
            \frac{1}{3}
            \end{pmatrix} \]
            
            The solution is:
            \[ U_{10} = 1,5 \, \text{V}, \quad U_{20} = 0,75 \, \text{V}, \quad U_{30} = 0,5 \, \text{V}. \]

        \item[\bf d)]
            Current \(I_4\) calculations:

            Current divided by \(R_4\) is:
            \[ I_4 = \frac{U_{20} - U_{30}}{R_4} \]
            Insertion:
            \[ I_4 = \frac{0,75 - 0,5}{2 \cdot 1} = \frac{0,25}{2} = 0,125 \, \text{A}. \]

	\end{itemize}
}
