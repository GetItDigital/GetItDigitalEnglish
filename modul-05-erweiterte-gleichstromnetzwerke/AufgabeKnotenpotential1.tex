\subsection{Nodal analysis method 1}
\Aufgabe{
    Given is the network of a real voltage source $U_\mathrm{q} = 3.6\ V$ with an internal resistance
    vo $R_\mathrm{i} = 0,2\ \Omega$.

    \begin{figure}[H]
        \resizebox{0.3\textwidth}{!}{%  
        \input{Tikz/src/A5Knot1.tex}
        }\centering                                                           
    \end{figure}

    Convert the specified real voltage source into a real current source.
    To do this, follow these steps:
        
    \begin{itemize}
            \item[\bf a)]
                Determine the conductance of the resistance value. 
            \item[\bf b)]
                Convert the real voltage source into a real current source.
            \item[\bf c)]
                Calculate the current value of the converted current source.
        \end{itemize}
    }

\Loesung{
	\begin{itemize}
        \item[\bf a)]
            Conductance:    \\
            \begin{eq}
                G_\mathrm{i} = \frac{1}{R_\mathrm{i}} = \frac{1}{0,2\ \Omega} = 5\ \mathrm{S}   \nonumber
            \end{eq}

		\item[\bf b)]
		    Current source:    \\
            \begin{figure}[H]
                \resizebox{0.3\textwidth}{!}{%  
                \input{Tikz/src/B5Knot1.tex}  
                }\centering                                                           
            \end{figure}

        \item[\bf c)]
            Current value \\
            \begin{eq}
                I_\mathrm{q} = U_\mathrm{q} \cdot G_\mathrm{i} = 3,6\ \mathrm{V} \cdot 5\ \mathrm{S} = 72\ \mathrm{A}     \nonumber
            \end{eq}

	\end{itemize}
}
