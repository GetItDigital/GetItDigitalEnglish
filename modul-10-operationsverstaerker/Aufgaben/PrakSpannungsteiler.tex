\subsection{Analysis of a loaded voltage divider\label{Aufg26}}
% Alt: Praktikumsaufgabe 12
\Aufgabe{
        Draw a circuit diagram consisting of a loaded voltage divider.  
        The component values are given as follows:
        
        \begin{itemize}
            \item Resistance: $R_1 = R_2 = 10\,k\Omega$
            \item Load resistance: $R_\mathrm{L} = 2{,}2\,\mathrm{k\Omega}$
            \item Supply voltage: $U_\mathrm{CC} = +15\,\mathrm{V}$
        \end{itemize}
        
        \begin{itemize}
            \item[a)] Calculate the output voltage $U_\mathrm{C}$ without load resistance.
            \item[b)] Calculate the output voltage $U_\mathrm{C}$ with the connected load resistance. $\mathrm{R_L}$.
            \item[c)] Explain why the voltage $U_\mathrm{C}$ drops across the load resistor.
        \end{itemize}
    
}

\Loesung{
    \begin{figure}[H]
        \centering
            \input{Tikz/src/LsgPrakSpannungsteiler.tex}
            \label{fig:LsgPrakSpannungsteiler}
    \end{figure}
    \begin{itemize}
        \item[a)] Calculation of the output voltage $U_\mathrm{C}$ without load resistance:
        \begin{align*}
            U_\mathrm{C} &= \frac{R_2}{R_1 + R_2} \cdot U_\mathrm{CC} \\
            U_\mathrm{C} &= \frac{10\,k\Omega}{10\,k\Omega + 10\,\mathrm{k\Omega}} \cdot 15\,\mathrm{V} \\
            U_\mathrm{C} &= \frac{1}{2} \cdot 15\,\mathrm{V} \\
            U_\mathrm{C} &= 7{,}5\,\mathrm{V}
        \end{align*}
        \item[b)] Calculation of the output voltage $\mathrm{U_C}$ with the connected load resistance $\mathrm{R_L}$:
        \begin{align*}
            U_\mathrm{C} &= \frac{R_2}{R_1 + R_2 + R_\mathrm{L}} \cdot U_\mathrm{CC} \\
            U_\mathrm{C} &= \frac{10\,\mathrm{k\Omega}}{10\,\mathrm{k\Omega} + 10\,\mathrm{k\Omega} + 2{,}2\,\mathrm{k\Omega}} \cdot 15\,\mathrm{V} \\
            U_\mathrm{C} &= \frac{10\,\mathrm{k\Omega}}{22\,\mathrm{k\Omega}} \cdot 15\,\mathrm{V} \\
            U_\mathrm{C} &= \frac{10}{22} \cdot 15\,\mathrm{V} \\
            U_\mathrm{C} &= \frac{150}{22}\,\mathrm{V} \\
            U_\mathrm{C} &\approx 6{,}82\,\mathrm{V}
        \end{align*}
        \item[c)] The voltage $U_\mathrm{C}$ decreases due to the load resistor $R_\mathrm{L}$, as it forms a voltage divider with resistors $R_1$ and $R_2$. 
        The load resistor $R_\mathrm{L}$ draws part of the voltage from the circuit, causing the output voltage $U_\mathrm{C}$ to drop.

    \end{itemize}
