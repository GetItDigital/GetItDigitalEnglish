\subsection{RMS value 1}
\Aufgabe{

    Given a sinusoidal voltage
    
    \begin{eq}
        u(t) = 150\ V \cdot \sin(100\pi t)  \nonumber
    \end{eq}

    where $t$ is given in seconds and the voltage $u(t)$ is given in volts.
    
    Calculate the following for this voltage:

    \begin{itemize}
        \item[\bf a)] The \textbf{arithmetic mean} $\bar{u}$ of the voltage over one period.
        \item[\bf b)] The \textbf{RMS value} $U_{RMS}$ of the voltage.
    \end{itemize}
}

\Loesung{
	\begin{itemize}

	\item[\bf a)]
        The arithmetic mean of a function u(t) over a period T is given by:
        \begin{eq}
            \bar{u} = \frac{1}{T} \int_0^T u(t) dt  \nonumber
        \end{eq}
        
        Since $u(t)$ represents a sine function and the sine wave is symmetrical over one period, 
        the positive and negative parts of the wave cancel each other out. Therefore, the arithmetic 
        mean for each sine wave is always zero:
        
        \begin{eq}
            \bar{u} = 0 \nonumber
        \end{eq}

    \item[\bf b)]
        The RMS value of a sinusoidal voltage $u(t)$ is calculated using the following formula:
        \begin{eq}
            U_{Eff} = \frac{\hat{U}}{\sqrt{2}}   \nonumber
        \end{eq}
        
        In this case, $\hat{U} = 150\ V$ is the amplitude of the voltage.
        
        \begin{eq}
            U_{Eff} = \frac{150\ V}{\sqrt{2}} \approx \frac{150\ V}{1.414} \approx 106.1\ V  \nonumber
        \end{eq}
	
\end{itemize}

}
