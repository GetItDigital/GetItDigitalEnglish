\subsection{Electric homogeneous field 1\label{ElHomFeld}}
\Aufgabe{
    A capacitor for the voltage range of $100 \ kV$ is to be realised with a dielectric in which a maximum field strength of $E = 30 \ \frac{kV}{cm}$ is permissible.
    Calculate the capacitance of the capacitor.\\

       \underline{Task section 1}:\\
		\phantom{text}\\

        The dielectric is first used in an (ideal) \underline{plate capacitor}.\\
        \begin{enumerate}[label=\alph*1)]
            \item Sketch the arrangement.
            \item Sketch the field lines in the arrangement.
            \item Where is the location of the maximum field strength?
            \item Sketch the course of the electric field strength from one capacitor plate to the other $\left| E(x) \right|$, parallel to the capacitor plates
            $\left| E_{Diel}(z) \right|$ and in the capacitor plates $\left| E_{Pl}(z) \right|$.
            \item Draw the 50\%- equipotential line.
            \item Calculate the minimum plate spacing $d_{Pl,min}$ for the above boundary conditions.           
        \end{enumerate} 

       \underline{Task section} 2:\\
	   \phantom{text}\\

        Now the \underline{dielectric between coaxial cylinders} is to be inserted.\\
        $\left(E_r(r)=\frac{U}{r\cdot\ln\left(\frac{a}{b}\right)};\ r_a: Outer radius; r_i: Radius \ of the \ inner cylinder\right)$
        \begin{enumerate}[label=\alph*2)]    
        \item Sketch the arrangement.
        \item Sketch the field lines.
        \item Where is the location of the maximum field strength?
        \item Sketch the course of the electric field strength $E_r(r)$ from $r=r_i$ to $r=r_a$.
        \item Draw the 50% equipotential line.    
        \item Calculate the minimum outer radius $r_a$ for $r_i = 1;\ 2;\ 3;\ 4;\ 5;\ 10 \ cm$ and
            the corresponding thickness of the dielectric $d_i$.
                        
        \end{enumerate}
}
